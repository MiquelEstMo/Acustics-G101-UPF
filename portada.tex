%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Portada -- variante "Burdeos" de una columna
% Franja vertical burdeos (taucolor) al estilo tau-class
%
% Lee los datos ya declarados en el preambulo de main.tex:
%   \doctype{...}        -> etiqueta superior
%   \title{...}          -> titulo principal
%   \subtitle{...}       -> subtitulo / descripcion breve
%   \portadaautores{...} -> lista de autores (si no existe se usa \leadauthor)
%   \organization{...}   -> institucion
%   \dates{...}          -> fecha o curso
%
% Logo opcional: colocar "logo.png" o "logo.pdf" en la raiz del proyecto.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\makeatletter
\begin{titlepage}
\newgeometry{margin=0pt}
\begin{tikzpicture}[remember picture, overlay]

  % Fondo blanco base
  \fill[white] (current page.south west) rectangle (current page.north east);

  % Franja vertical (taucolor) a la izquierda
  \fill[taucolor] (current page.south west) rectangle
      ([xshift=3.2cm]current page.south west |- current page.north west);

  % Linea fina blanca separando la franja del resto
  \draw[white, line width=0.4pt]
      ([xshift=3.2cm]current page.south west) --
      ([xshift=3.2cm]current page.north west);

  % Texto vertical decorativo dentro de la franja (organizacion)
  \node[rotate=90, anchor=center, white, font=\sffamily\small]
      at ([xshift=0.95cm]current page.west)
      {\ifx\@organization\undefined\else\MakeUppercase{\@organization}\fi};

  % Logo dentro de la franja (opcional)
  \IfFileExists{logo.png}{
    \node[anchor=north] at ([xshift=1.6cm, yshift=-1.9cm]current page.north west)
        {\includegraphics[width=2.5cm]{logo.png}};
  }{
    \IfFileExists{logo.pdf}{
      \node[anchor=north] at ([xshift=1.6cm, yshift=-1.9cm]current page.north west)
          {\includegraphics[width=2.5cm]{logo.pdf}};
    }{}
  }

  % Bloque de texto principal (fuera de la franja)
  \node[anchor=north west, align=left, text width=13.5cm]
      at ([xshift=4.3cm, yshift=-7.8cm]current page.north west) {
      {\ifx\@doctype\undefined\else\sffamily\footnotesize\color{taugray}\MakeUppercase{\@doctype}\\[10pt]\fi}
      {\fontsize{30}{34}\selectfont\bfseries\color{taucolor}\@title}\\[6pt]
      {\ifx\@subtitle\undefined\else\sffamily\large\color{black}\@subtitle\fi}
  };

  % Linea horizontal fina bajo el titulo
  \draw[taucolor, line width=0.8pt]
      ([xshift=4.3cm, yshift=-12.2cm]current page.north west) --
      ([xshift=16.3cm, yshift=-12.2cm]current page.north west);

  % Bloque de autores / organizacion / fecha (parte inferior)
  \node[anchor=south west, align=left, text width=13.5cm]
      at ([xshift=4.3cm, yshift=2.8cm]current page.south west) {
      {\ifx\@portadaautores\undefined
         \ifx\@leadauthor\undefined\else\sffamily\bfseries\color{black}\@leadauthor\\[2pt]\fi
       \else
         \sffamily\color{black}\@portadaautores\\[8pt]
       \fi}
      {\ifx\@organization\undefined\else\sffamily\itshape\small\color{taugray}\@organization\\[10pt]\fi}
      {\sffamily\footnotesize\color{taugray}\ifx\@dates\undefined\today\else\@dates\fi}
  };

\end{tikzpicture}
\end{titlepage}
\restoregeometry
\makeatother
